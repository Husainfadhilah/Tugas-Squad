## 1. Kode JavaScript Sederhana:
function hitungTotalBayar(totalBelanja, jarakKm) {
  let diskon = 0;
  let ongkir = 0;

  // 1. Hitung Diskon
  if (totalBelanja >= 100000) {
    diskon = totalBelanja * 0.10; // Diskon 10%
    
    // Batas maksimal diskon Rp 20.000
    if (diskon > 20000) {
      diskon = 20000;
    }
  } else {
    diskon = 0;
  }

  // 2. Hitung Biaya Ongkir
  if (jarakKm <= 2) {
    ongkir = 0; // Gratis ongkir untuk jarak <= 2 km
  } else {
    ongkir = 5000; // Biaya ongkir untuk jarak > 2 km
  }

  // 3. Hitung Total Akhir
  let totalAkhir = totalBelanja - diskon + ongkir;

  return {
    totalBelanja: totalBelanja,
    jarakKm: jarakKm,
    diskon: diskon,
    ongkir: ongkir,
    totalAkhir: totalAkhir
  };
}

// Uji Coba sesuai Tabel Dry Run
